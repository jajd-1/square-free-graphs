import numpy as np


def construct_new_graph(A, rng=None):
    """Convert the supplied graph into a dense C4-free graph by repair and beam search."""
    if A.dtype != np.uint8:
        raise TypeError("A must have dtype np.uint8")
    if A.ndim != 2 or A.shape[0] != A.shape[1]:
        raise ValueError("A must be a square adjacency matrix")

    if rng is None:
        rng = np.random.default_rng()

    n = A.shape[0]
    source = ((A != 0) | (A.T != 0)).astype(np.uint8)
    np.fill_diagonal(source, 0)

    def common_counts(H):
        X = H.astype(np.int16, copy=False)
        return X @ X

    def valid(H):
        C = common_counts(H)
        return not np.any(np.triu(C > 1, 1))

    def repair(H):
        """
        Remove edges covering many simultaneous C4 violations.  This always
        begins from the supplied state or a state derived from it.
        """
        H = H.copy()
        max_steps = n * (n - 1) // 2

        for _ in range(max_steps):
            C = common_counts(H)
            bad = C > 1
            np.fill_diagonal(bad, False)

            if not np.any(bad):
                return H

            # An edge uv receives support from violating pairs for which the
            # edge is one of the two paths through a common neighbor.
            support = bad.astype(np.int16) @ H.astype(np.int16)
            score = (support + support.T) * H
            best = int(score.max())

            if best > 0:
                choices = np.argwhere(np.triu(score == best, 1))
                if len(choices) > 1:
                    deg = H.sum(axis=1)
                    load = deg[choices[:, 0]] + deg[choices[:, 1]]
                    choices = choices[load == load.max()]
                u, v = choices[int(rng.integers(len(choices)))]
                u, v = int(u), int(v)
            else:
                pairs = np.argwhere(np.triu(bad, 1))
                i, j = pairs[int(rng.integers(len(pairs)))]
                common = np.flatnonzero(H[i] & H[j])
                w = int(common[int(rng.integers(len(common)))])
                if H[i].sum() >= H[j].sum():
                    u, v = int(i), w
                else:
                    u, v = int(j), w

            H[u, v] = 0
            H[v, u] = 0

        # Conservative final verifier.
        for _ in range(max_steps):
            C = common_counts(H)
            pairs = np.argwhere(np.triu(C > 1, 1))
            if len(pairs) == 0:
                break
            i, j = map(int, pairs[0])
            w = int(np.flatnonzero(H[i] & H[j])[0])
            H[i, w] = 0
            H[w, i] = 0

        return H

    def to_masks(H):
        masks = [0] * n
        for u in range(n):
            m = 0
            for v in np.flatnonzero(H[u]):
                m |= 1 << int(v)
            masks[u] = m
        return masks

    def from_masks(masks):
        H = np.zeros((n, n), dtype=np.uint8)
        for u, m in enumerate(masks):
            x = m
            while x:
                b = x & -x
                v = b.bit_length() - 1
                H[u, v] = 1
                x ^= b
        np.fill_diagonal(H, 0)
        return H

    def legal_add(masks, u, v):
        """
        uv is legal precisely when no edge joins N(u) and N(v).  Such an edge
        would complete a length-three u--x--y--v path into a C4.
        """
        nv = masks[v]
        x = masks[u]
        while x:
            b = x & -x
            w = b.bit_length() - 1
            if masks[w] & nv:
                return False
            x ^= b
        return True

    def saturate(masks, degrees):
        """
        Repeatedly choose a safe missing edge.  Low degree endpoints are
        preferred, while the degree-product term avoids making a few
        high-degree vertices consume too much of the available pair budget.
        """
        max_add = n * (n - 1) // 2

        for _ in range(max_add):
            best_key = None
            tied = []

            for u in range(n - 1):
                mu = masks[u]
                du = degrees[u]
                for v in range(u + 1, n):
                    if (mu >> v) & 1:
                        continue
                    if not legal_add(masks, u, v):
                        continue

                    dv = degrees[v]
                    key = (
                        du + dv,
                        du * dv,
                        abs(du - dv),
                        max(du, dv),
                    )
                    if best_key is None or key < best_key:
                        best_key = key
                        tied = [(u, v)]
                    elif key == best_key:
                        tied.append((u, v))

            if not tied:
                break

            u, v = tied[int(rng.integers(len(tied)))]
            masks[u] |= 1 << v
            masks[v] |= 1 << u
            degrees[u] += 1
            degrees[v] += 1

        return masks, degrees

    def maximize_from_matrix(H):
        masks = to_masks(H)
        degrees = [m.bit_count() for m in masks]
        masks, degrees = saturate(masks, degrees)
        return masks, degrees

    def state_key(masks):
        # A cheap signature used only to avoid retaining exact duplicate beam states.
        return tuple(masks)

    def edge_list(masks, degrees):
        items = []
        for u in range(n - 1):
            x = masks[u] >> (u + 1)
            v = u + 1
            while x:
                if x & 1:
                    # This estimates the number of cross-neighborhood pairs
                    # constrained by the edge and is useful for ruin selection.
                    value = (degrees[u] - 1) * (degrees[v] - 1)
                    value += degrees[u] + degrees[v]
                    items.append((value, u, v))
                x >>= 1
                v += 1
        return items

    # Several independently tie-broken repairs preserve input information but
    # avoid relying on a single deletion trajectory.
    beam = []
    seen = set()
    best_masks = None
    best_degrees = None
    best_edges = -1

    for _ in range(4):
        repaired = repair(source)
        masks, degrees = maximize_from_matrix(repaired)
        H = from_masks(masks)

        if not valid(H):
            repaired = repair(H)
            masks, degrees = maximize_from_matrix(repaired)
            H = from_masks(masks)

        count = int(H.sum() // 2)
        sig = state_key(masks)
        if sig not in seen:
            seen.add(sig)
            beam.append((count, masks[:], degrees[:]))

        if count > best_edges:
            best_edges = count
            best_masks = masks[:]
            best_degrees = degrees[:]

    beam.sort(key=lambda z: z[0], reverse=True)
    beam = beam[:3]

    # Beam ruin-and-refill.  Deletions preserve validity; all refilled edges
    # pass the bitset C4 test.  A mixture of shallow and deeper ruins explores
    # exchanges inaccessible to ordinary single-edge maximality.
    for round_id in range(6):
        proposals = []

        for rank, (_, base_masks, base_degrees) in enumerate(beam):
            attempts = 2 if rank == 0 else 1

            for attempt in range(attempts):
                masks = base_masks[:]
                degrees = base_degrees[:]
                edges = edge_list(masks, degrees)
                if not edges:
                    continue

                edges.sort(reverse=True)
                remove_count = 3 + ((round_id + attempt + rank) % 5)

                for r in range(remove_count):
                    available = [
                        item for item in edges
                        if (masks[item[1]] >> item[2]) & 1
                    ]
                    if not available:
                        break

                    # Most moves target strongly blocking edges, but a broad
                    # pool periodically introduces distinct structural changes.
                    if (round_id + r + attempt) % 4 == 0:
                        pool_size = max(1, min(len(available), len(available) // 2))
                    else:
                        pool_size = max(1, min(len(available), len(available) // 4 + 2))

                    _, u, v = available[int(rng.integers(pool_size))]
                    masks[u] &= ~(1 << v)
                    masks[v] &= ~(1 << u)
                    degrees[u] -= 1
                    degrees[v] -= 1

                masks, degrees = saturate(masks, degrees)
                H = from_masks(masks)

                # Explicit invariant check after each large local-search phase.
                if not valid(H):
                    H = repair(H)
                    masks, degrees = maximize_from_matrix(H)
                    H = from_masks(masks)

                count = int(H.sum() // 2)
                proposals.append((count, masks[:], degrees[:]))

                if count > best_edges:
                    best_edges = count
                    best_masks = masks[:]
                    best_degrees = degrees[:]

        # Preserve a small diverse elite beam, including previous states.
        combined = beam + proposals
        combined.sort(key=lambda z: z[0], reverse=True)

        next_beam = []
        seen = set()
        for item in combined:
            sig = state_key(item[1])
            if sig not in seen:
                seen.add(sig)
                next_beam.append(item)
            if len(next_beam) >= 3:
                break

        if next_beam:
            beam = next_beam

    result = from_masks(best_masks)

    # Mandatory final verification.  This is normally already true because
    # repair and safe saturation preserve the pair-common-neighbor invariant.
    if not valid(result):
        result = repair(result)
        masks, _ = maximize_from_matrix(result)
        result = from_masks(masks)

    return result.astype(np.uint8, copy=False)
