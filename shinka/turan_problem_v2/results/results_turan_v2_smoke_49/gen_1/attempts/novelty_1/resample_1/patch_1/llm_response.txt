<DESCRIPTION>
This approach keeps the input graph as the initial state, repairs it using an incrementally maintained common-neighbor matrix, and then performs degree-balanced saturation. Unlike the original one-pass repair, every deletion is chosen to eliminate many current C4 violations. After reaching a valid C4-free graph, it repeatedly adds the legal non-edge with the best degree-balancing score.

It then applies several bounded “destroy and refill” local-search rounds: remove a few highly blocking edges, greedily refill the resulting gaps, and retain the best valid graph found. All additions are checked through the invariant that every off-diagonal entry of the common-neighbor matrix remains at most one. A final explicit validation-and-repair pass is included for safety.

# EVOLVE-BLOCK-START
import numpy as np

def construct_new_graph(A, rng=None):
    """Repair, saturate, and locally improve a C4-free subgraph of A."""
    if A.dtype != np.uint8:
        raise TypeError("A must have dtype np.uint8")

    if rng is None:
        rng = np.random.default_rng()

    n = A.shape[0]

    # Work from the supplied graph itself.  Symmetrizing is conservative for
    # malformed inputs and the diagonal is never an edge.
    G = ((A != 0) | (A.T != 0)).astype(np.uint8)
    np.fill_diagonal(G, 0)

    def common_matrix(H):
        return (H.astype(np.int16) @ H.astype(np.int16))

    def remove_edge(H, C, u, v):
        """Delete uv and update the common-neighbor counts exactly."""
        nu = np.flatnonzero(H[u])
        nv = np.flatnonzero(H[v])
        nu = nu[nu != v]
        nv = nv[nv != u]

        if nv.size:
            C[u, nv] -= 1
            C[nv, u] -= 1
        if nu.size:
            C[v, nu] -= 1
            C[nu, v] -= 1

        H[u, v] = 0
        H[v, u] = 0

    def add_edge(H, C, u, v):
        """Add a previously checked legal edge and update counts exactly."""
        nu = np.flatnonzero(H[u])
        nv = np.flatnonzero(H[v])

        if nv.size:
            C[u, nv] += 1
            C[nv, u] += 1
        if nu.size:
            C[v, nu] += 1
            C[nu, v] += 1

        H[u, v] = 1
        H[v, u] = 1

    def repair(H):
        """Delete edges from H until every pair has at most one common neighbor."""
        C = common_matrix(H)
        limit = n * n * 3

        for _ in range(limit):
            bad = np.argwhere(np.triu(C > 1, 1))
            if bad.size == 0:
                break

            # Repair a worst violation first.
            vals = C[bad[:, 0], bad[:, 1]]
            p = int(np.argmax(vals))
            i, j = int(bad[p, 0]), int(bad[p, 1])

            common = np.flatnonzero(H[i] & H[j])
            if common.size < 2:
                # Defensive recomputation in case of any unusual input issue.
                C = common_matrix(H)
                continue

            candidates = []
            scores = []

            # Deleting i-x or j-x resolves the chosen violation. Prefer the
            # deletion that simultaneously decreases the most other violations.
            for x in common:
                x = int(x)

                nx = np.flatnonzero(H[x])
                ni = np.flatnonzero(H[i])
                affected1 = nx[nx != i]
                affected2 = ni[ni != x]
                score_ix = 0
                if affected1.size:
                    score_ix += int(np.count_nonzero(C[i, affected1] > 1))
                if affected2.size:
                    score_ix += int(np.count_nonzero(C[x, affected2] > 1))
                candidates.append((i, x))
                scores.append(score_ix)

                nj = np.flatnonzero(H[j])
                affected1 = nx[nx != j]
                affected2 = nj[nj != x]
                score_jx = 0
                if affected1.size:
                    score_jx += int(np.count_nonzero(C[j, affected1] > 1))
                if affected2.size:
                    score_jx += int(np.count_nonzero(C[x, affected2] > 1))
                candidates.append((j, x))
                scores.append(score_jx)

            scores = np.asarray(scores)
            best_score = scores.max()
            tied = np.flatnonzero(scores == best_score)
            choice = int(tied[int(rng.integers(len(tied)))])
            u, v = candidates[choice]
            remove_edge(H, C, u, v)

        # Explicitly verify the invariant; this fallback is intentionally
        # simple and only executes if an incremental update was ever suspect.
        C = common_matrix(H)
        for _ in range(n * n):
            bad = np.argwhere(np.triu(C > 1, 1))
            if bad.size == 0:
                break
            i, j = map(int, bad[0])
            shared = np.flatnonzero(H[i] & H[j])
            if shared.size == 0:
                C = common_matrix(H)
                continue
            x = int(shared[0])
            remove_edge(H, C, i, x)

        return C

    def legal(C, H, u, v):
        # Adding uv creates a C4 exactly when some neighbor x of u already
        # has a common neighbor with v (a length-three u--x--y--v path).
        nu = np.flatnonzero(H[u])
        if nu.size and np.any(C[nu, v] > 0):
            return False
        return True

    def saturate(H, C):
        """Repeatedly add a legal edge, favoring balanced low-degree endpoints."""
        max_steps = n * (n - 1) // 2
        for _ in range(max_steps):
            deg = H.sum(axis=1).astype(np.int16)
            best_u = -1
            best_v = -1
            best_score = None

            for u in range(n - 1):
                for v in range(u + 1, n):
                    if H[u, v]:
                        continue
                    if not legal(C, H, u, v):
                        continue

                    # Near-regular graphs are usually much denser in the
                    # C4-free regime. Small random noise diversifies ties.
                    score = (
                        8 * int(deg[u] + deg[v])
                        + abs(int(deg[u]) - int(deg[v]))
                        + float(rng.random()) * 0.15
                    )
                    if best_score is None or score < best_score:
                        best_score = score
                        best_u, best_v = u, v

            if best_u < 0:
                break
            add_edge(H, C, best_u, best_v)

        return C

    # Repair the actual input graph, then greedily augment it.
    C = repair(G)
    C = saturate(G, C)

    best_G = G.copy()
    best_edges = int(G.sum() // 2)

    # Bounded ruin-and-recreate local search.  Removing edges incident to
    # high-degree vertices tends to release many blocked additions.
    for _round in range(5):
        H = best_G.copy()
        Cwork = common_matrix(H)
        deg = H.sum(axis=1).astype(np.int16)

        edges = np.argwhere(np.triu(H, 1))
        if edges.size == 0:
            break

        values = []
        for u, v in edges:
            u, v = int(u), int(v)
            values.append((int(deg[u] - 1) * int(deg[v] - 1), u, v))
        values.sort(reverse=True)

        # Delete a small diverse set of highly constraining edges.
        removed = 0
        used = set()
        for _, u, v in values:
            if removed >= 3:
                break
            if u in used and v in used:
                continue
            remove_edge(H, Cwork, u, v)
            used.add(u)
            used.add(v)
            removed += 1

        Cwork = saturate(H, Cwork)

        # Mandatory final invariant check before accepting a local-search state.
        verify = common_matrix(H)
        if np.any(np.triu(verify > 1, 1)):
            verify = repair(H)
            verify = saturate(H, verify)

        edge_count = int(H.sum() // 2)
        if edge_count >= best_edges:
            best_G = H
            best_edges = edge_count

    # Final explicit verification and conservative repair if needed.
    final_C = common_matrix(best_G)
    if np.any(np.triu(final_C > 1, 1)):
        final_C = repair(best_G)
        saturate(best_G, final_C)

    return best_G.astype(np.uint8)

# EVOLVE-BLOCK-END


def run_graph_construction(A, rng=None):
    """Run the graph construction algorithm on A"""
    return construct_new_graph(A=A, rng=rng)
