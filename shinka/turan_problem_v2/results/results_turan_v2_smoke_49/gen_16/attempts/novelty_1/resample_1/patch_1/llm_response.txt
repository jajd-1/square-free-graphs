<DESCRIPTION>
The current approach uses a six-state archive, five input repairs, loss weight 4, load weight 1, and 18 destructive unlock kicks. The new configuration increases structural diversity at initialization by using seven input-derived repairs and three complementary saturation modes: strong future-loss preservation, stronger degree balancing, and a degree-cap mode that discourages early hubs above the typical degree-6/7 extremal range.

The archive is enlarged to eight states, while kicks use adaptive edge-unlocking scores combined with degree pressure. Kicks alternate between distributed removals and concentrated removals, and refill modes vary rather than repeatedly using one deterministic basin. All additions explicitly use the length-three-path legality condition, all deletion phases are monotone safe, and final validation checks the common-neighbor invariant.

"""Balanced archive search with unlock-based C4-free local improvement."""

import numpy as np


def construct_new_graph(A, rng=None):
    """
    Directly edit the supplied adjacency matrix into a dense C4-free graph.

    The procedure:
      * greedily repairs several differently ordered subsets of input edges;
      * completes each repair using safe additions with several scoring modes;
      * maintains an archive of high-quality saturated C4-free states;
      * applies bounded unlock-oriented delete/refill kicks.
    """
    if A.dtype != np.uint8:
        raise TypeError("A must have dtype np.uint8")

    if rng is None:
        rng = np.random.default_rng()

    n = A.shape[0]
    iu, ju = np.triu_indices(n, 1)
    m = iu.size

    source = ((A != 0) | (A.T != 0)).astype(np.uint8)
    np.fill_diagonal(source, 0)
    source_upper = source[iu, ju].astype(bool)
    source_degree = source.sum(axis=1).astype(np.int32)
    source_edges = np.flatnonzero(source_upper)

    ARCHIVE_LIMIT = 8
    TARGET_DEGREE = 6

    def edge_count(B):
        return int(B.sum() // 2)

    def valid(B):
        X = B.astype(np.int16)
        common = X @ X
        np.fill_diagonal(common, 0)
        return not np.any(common > 1)

    def insertion_legal(B, u, v):
        nu = np.flatnonzero(B[u])
        nv = np.flatnonzero(B[v])
        if nu.size == 0 or nv.size == 0:
            return True
        return not np.any(B[np.ix_(nu, nv)])

    def legal_matrix(B):
        X = B.astype(np.int16)
        p3 = (X @ X) @ X
        legal = (B == 0) & (p3 == 0)
        np.fill_diagonal(legal, False)
        return legal, p3

    def saturate(B, mode=0, banned=None):
        """
        Add all possible safe edges.  Modes alter only tie-breaking/scoring:
        mode 0: future-loss dominated;
        mode 1: stronger regularity pressure;
        mode 2: explicit degree-six/seven cap pressure.
        """
        B = B.copy()
        forbidden = (
            np.zeros(m, dtype=bool) if banned is None else banned.copy()
        )

        for _ in range(m):
            legal, _ = legal_matrix(B)
            available = np.flatnonzero(legal[iu, ju] & ~forbidden)

            if available.size == 0 and np.any(forbidden):
                forbidden[:] = False
                available = np.flatnonzero(legal[iu, ju])

            if available.size == 0:
                break

            X = B.astype(np.int16)
            L = legal.astype(np.int16)
            loss_matrix = (X @ L) @ X
            degree = B.sum(axis=1).astype(np.int32)

            au = iu[available]
            av = ju[available]
            loss = loss_matrix[au, av].astype(np.int32)
            load = degree[au] + degree[av]

            if mode == 0:
                score = 5 * loss + load
                width = 1
            elif mode == 1:
                score = 3 * loss + 2 * load
                width = 2
            else:
                new_u = degree[au] + 1
                new_v = degree[av] + 1
                excess = (
                    np.maximum(new_u - (TARGET_DEGREE + 1), 0) +
                    np.maximum(new_v - (TARGET_DEGREE + 1), 0)
                )
                deviation = (
                    np.abs(new_u - TARGET_DEGREE) +
                    np.abs(new_v - TARGET_DEGREE)
                )
                score = 4 * loss + 3 * excess + deviation
                width = 1

            minimum = score.min()
            band = available[score <= minimum + width]
            if band.size == 0:
                band = available[score == minimum]

            input_band = band[source_upper[band]]
            if input_band.size and rng.random() < 0.30:
                band = input_band

            chosen = int(band[int(rng.integers(band.size))])
            u = int(iu[chosen])
            v = int(ju[chosen])

            # Explicit local invariant check before every insertion.
            if insertion_legal(B, u, v):
                B[u, v] = 1
                B[v, u] = 1

        return B

    def repair_input(style):
        """
        Construct a valid edit state by retaining source relationships in an
        input-dependent order.  The later completion phase adds missing edges.
        """
        B = np.zeros((n, n), dtype=np.uint8)
        if source_edges.size == 0:
            return B

        du = source_degree[iu[source_edges]]
        dv = source_degree[ju[source_edges]]
        load = du + dv
        product = du * dv
        imbalance = np.abs(du - dv)
        noise = rng.random(source_edges.size)

        if style == 0:
            order = source_edges[np.lexsort((noise, load))]
        elif style == 1:
            order = source_edges[np.lexsort((noise, -load))]
        elif style == 2:
            order = source_edges[np.lexsort((noise, product))]
        elif style == 3:
            order = source_edges[np.lexsort((noise, -product))]
        elif style == 4:
            order = source_edges[np.lexsort((noise, imbalance))]
        elif style == 5:
            order = source_edges[np.lexsort((noise, -imbalance))]
        else:
            order = source_edges[rng.permutation(source_edges.size)]

        for e in order:
            u = int(iu[e])
            v = int(ju[e])
            if insertion_legal(B, u, v):
                B[u, v] = 1
                B[v, u] = 1

        return B

    def archive_insert(archive, candidate):
        if not valid(candidate):
            return archive

        signature = candidate[iu, ju].tobytes()
        count = edge_count(candidate)

        for old_count, _, old_signature in archive:
            if old_signature == signature:
                return archive

        archive.append((count, candidate.copy(), signature))
        archive.sort(key=lambda item: item[0], reverse=True)
        return archive[:ARCHIVE_LIMIT]

    def unlock_scores(B, present):
        """
        Score present edges by the number of unique length-three obstructions
        they destroy when removed.  These are precisely the missing pairs most
        likely to become legal after the deletion.
        """
        _, p3 = legal_matrix(B)
        degree = B.sum(axis=1).astype(np.int32)
        scores = np.zeros(present.size, dtype=np.int32)

        for pos, e in enumerate(present):
            a = int(iu[e])
            b = int(ju[e])
            na = np.flatnonzero(B[a])
            nb = np.flatnonzero(B[b])

            total = 0
            for x in na:
                if x == b:
                    continue
                for y in nb:
                    if y == a or y == x:
                        continue
                    if B[x, y] == 0 and p3[x, y] == 1:
                        total += 1

            # The opposite orientation may unlock distinct endpoint pairs.
            for x in nb:
                if x == a:
                    continue
                for y in na:
                    if y == b or y == x:
                        continue
                    if B[x, y] == 0 and p3[x, y] == 1:
                        total += 1

            scores[pos] = 14 * total + degree[a] + degree[b]

        return scores

    archive = []

    # Seven differently repaired input states, each completed with a rotating
    # structural scoring mode.
    for style in range(7):
        state = repair_input(style)
        if not valid(state):
            continue

        state = saturate(state, mode=style % 3)
        if valid(state):
            archive = archive_insert(archive, state)

    if not archive:
        fallback = np.zeros((n, n), dtype=np.uint8)
        fallback = saturate(fallback, mode=0)
        archive = archive_insert(archive, fallback)

    # (edges removed, unlock-score percentile, refill mode)
    kick_schedule = (
        (2, 76, 1), (2, 70, 2), (2, 64, 0), (2, 58, 1),
        (3, 68, 2), (3, 62, 0), (3, 56, 1), (3, 50, 2),
        (4, 60, 0), (4, 52, 1), (4, 44, 2),
        (3, 46, 0), (2, 52, 2), (3, 40, 1),
        (4, 36, 0), (2, 44, 1), (3, 34, 2),
        (4, 30, 1),
    )

    for trial, (remove_count, percentile, mode) in enumerate(kick_schedule):
        if len(archive) == 1 or rng.random() < 0.60:
            base = archive[0][1]
        else:
            # Most alternate choices are elite, but retain some diversity.
            rank_limit = min(len(archive), 4 if rng.random() < 0.75 else len(archive))
            base = archive[int(rng.integers(rank_limit))][1]

        current = base.copy()
        present = np.flatnonzero(current[iu, ju])
        if present.size < remove_count:
            continue

        scores = unlock_scores(current, present)
        cutoff = np.percentile(scores, percentile)
        pool = present[scores >= cutoff]
        if pool.size < remove_count:
            pool = present

        selected = []
        used = set()

        for pick in range(remove_count):
            choices = pool[~np.isin(pool, selected)]
            if choices.size == 0:
                choices = present[~np.isin(present, selected)]

            # Distributed deletions often unlock independent regions.  Every
            # fourth kick permits concentration to dismantle a local gadget.
            if pick > 0 and trial % 4 != 0:
                disjoint = np.array(
                    [
                        e for e in choices
                        if int(iu[e]) not in used and int(ju[e]) not in used
                    ],
                    dtype=np.int64,
                )
                if disjoint.size:
                    choices = disjoint

            chosen = int(choices[int(rng.integers(choices.size))])
            selected.append(chosen)
            used.add(int(iu[chosen]))
            used.add(int(ju[chosen]))

        banned = np.zeros(m, dtype=bool)
        for e in selected:
            u = int(iu[e])
            v = int(ju[e])
            current[u, v] = 0
            current[v, u] = 0
            banned[e] = True

        # Deletion preserves C4-freeness; retain an explicit verification.
        if not valid(current):
            continue

        candidate = saturate(current, mode=mode, banned=banned)
        if valid(candidate):
            archive = archive_insert(archive, candidate)

    best = archive[0][1].copy()

    # Explicit final invariant verification and conservative repair fallback.
    repair_budget = n * n
    for _ in range(repair_budget):
        X = best.astype(np.int16)
        common = X @ X
        np.fill_diagonal(common, 0)
        bad = np.argwhere(np.triu(common > 1, 1))
        if bad.size == 0:
            break

        u, v = map(int, bad[0])
        shared = np.flatnonzero(best[u] & best[v])
        if shared.size == 0:
            continue

        degree = best.sum(axis=1)
        x = int(shared[np.argmax(degree[shared])])
        if degree[u] >= degree[v]:
            best[u, x] = 0
            best[x, u] = 0
        else:
            best[v, x] = 0
            best[x, v] = 0

    if valid(best):
        best = saturate(best, mode=0)

    # Strict monotone safety fallback.
    if not valid(best):
        for _ in range(repair_budget):
            X = best.astype(np.int16)
            common = X @ X
            np.fill_diagonal(common, 0)
            bad = np.argwhere(np.triu(common > 1, 1))
            if bad.size == 0:
                break

            u, v = map(int, bad[0])
            shared = np.flatnonzero(best[u] & best[v])
            if shared.size:
                x = int(shared[0])
                best[u, x] = 0
                best[x, u] = 0

    np.fill_diagonal(best, 0)
    return best.astype(np.uint8)


# EVOLVE-BLOCK-END


# The following code remains fixed (not evolved)

def run_graph_construction(A, rng=None):
    """Run the graph construction algorithm on A"""
    return construct_new_graph(A=A, rng=rng)
